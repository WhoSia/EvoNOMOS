func observe(wrap func(http.Handler) http.Handler, raw, body string) int {
 called := 0
 h := wrap(http.HandlerFunc(func(w http.ResponseWriter,r *http.Request){called++;w.WriteHeader(204)}))
 req:=httptest.NewRequest("POST","/",strings.NewReader(body))
 if raw!="" {req.Header.Set("Content-Type",raw)}
 rec:=httptest.NewRecorder();h.ServeHTTP(rec,req)
 if (rec.Code==204 && called!=1)||(rec.Code==415 && called!=0)||(rec.Code!=204 && rec.Code!=415){panic("invalid status/call count")}
 return rec.Code
}
