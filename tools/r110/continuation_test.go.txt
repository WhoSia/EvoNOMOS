package mem_test

import (
 "bytes"
 "testing"
 "google.golang.org/grpc/mem"
)

// The protected first buffer stays live; each subsequent copy is freed before the next.
func TestContinuationQuotient(t *testing.T) {
 sizes:=[]int{1025,2048,4096}
 cases,checks:=0,0
 for _,prime:=range sizes {for _,first:=range sizes {for word:=0;word<216;word++ {
  p:=&pool{mode:0}
  primed:=mem.Copy(bytes.Repeat([]byte{3},prime),p);primed.Free()
  want:=bytes.Repeat([]byte{1},first);a:=mem.Copy(want,p)
  high,attached,failed:=max(prime,first),true,false
  observedFailed:=false;n:=word
  for step:=0;step<3;step++ {
   symbol:=n%6;n/=6;size:=sizes[symbol/2];value:=byte(1+symbol%2)
   if size>high {high=size;attached=false} else if attached&&value!=1 {failed=true}
   b:=mem.Copy(bytes.Repeat([]byte{value},size),p)
   observedFailed=observedFailed||!bytes.Equal(a.ReadOnlyData(),want)
   if observedFailed!=failed {t.Fatalf("prime=%d first=%d word=%d step=%d",prime,first,word,step)}
   b.Free();checks++
  }
  a.Free();if p.puts!=5 {t.Fatal("accounting",p.puts)};cases++
 }}}
 if cases!=1944||checks!=5832 {t.Fatal(cases,checks)}
 t.Logf("cases=%d prefix_checks=%d",cases,checks)
}
