package mem_test

import (
 "bytes"
 "testing"
 "google.golang.org/grpc/mem"
)

// A value contract is weaker than a requirement for exclusive ownership.
func TestObservationSampling(t *testing.T) {
 cases, bad, falseAlarm, restored := 0, 0, 0, 0
 for a:=byte(1);a<=2;a++ {for b:=byte(1);b<=2;b++ {for c:=byte(1);c<=2;c++ {
  for mask:=1;mask<=3;mask++ {
   p:=&pool{mode:0}
   initial:=bytes.Repeat([]byte{a},2048)
   first:=mem.Copy(initial,p)
   second:=mem.Copy(bytes.Repeat([]byte{b},2048),p)
   if &first.ReadOnlyData()[0]!=&second.ReadOnlyData()[0] {t.Fatal("not same region")}
   changed2:=!bytes.Equal(initial,first.ReadOnlyData())
   third:=mem.Copy(bytes.Repeat([]byte{c},2048),p)
   if &first.ReadOnlyData()[0]!=&third.ReadOnlyData()[0] {t.Fatal("not same region")}
   changed3:=!bytes.Equal(initial,first.ReadOnlyData())
   observed:=mask&1!=0&&changed2||mask&2!=0&&changed3
   predicted:=mask&1!=0&&a!=b||mask&2!=0&&a!=c
   if observed!=predicted {t.Fatalf("%d %d %d mask %d",a,b,c,mask)}
   third.Free();second.Free();first.Free()
   if p.puts!=3 {t.Fatal("accounting")}
   cases++;if observed {bad++} else {falseAlarm++}
   if mask==2&&changed2&&!changed3 {restored++}
  }
 }}}
 if cases!=24||bad!=14||falseAlarm!=10||restored!=2 {t.Fatal(cases,bad,falseAlarm,restored)}
 t.Logf("cases=%d sampled_violations=%d overlap_false_positives=%d restored_before_final_sample=%d",cases,bad,falseAlarm,restored)
}
